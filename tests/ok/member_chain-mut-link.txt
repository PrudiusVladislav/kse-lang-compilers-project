struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Circle
{
    Point mut center
    i32 radius
}
Point p{3, 4}
Circle mut c{p, 2}
i32 n{c.center.shift(10)}
print(n)
print(p.x)
exit c.center.x
